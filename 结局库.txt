# -------------------- 结局生成器（优化版） --------------------
def generate_ending(player: Dict[str, Any], max_results: int = 3) -> List[Dict[str, str]]:
    """
    根据玩家最终状态生成正能量结局列表。
    按优先级降序排列，返回前 max_results 个结局（去重）。
    如果玩家表现优秀，可能返回多个结局；若表现普通则返回默认结局。
    :param player: 玩家数据字典，包含属性、标记等
    :param max_results: 最多返回的结局数量
    :return: 结局字典列表，每个包含标题和描述
    """
    matched = []
    for ending in ENDING_LIBRARY:
        if _check_ending_condition(player, ending["condition"]):
            matched.append(ending)

    # 按优先级降序排序
    matched.sort(key=lambda x: x["priority"], reverse=True)

    # 去重（通常不会重复，但安全）
    seen = set()
    result = []
    for e in matched:
        if e["id"] not in seen:
            result.append({"title": e["title"], "description": e["description"]})
            seen.add(e["id"])
        if len(result) >= max_results:
            break

    # 如果结果为空（理论上默认结局会兜底），返回默认
    if not result:
        result.append({
            "title": "平安完成军训",
            "description": "你顺利完成了军训，虽然过程中没有突出的表现，但你的坚持和努力值得肯定。军训的结束是新的开始，愿你把军训中的收获带到未来的生活中。"
        })
    return result


def _check_ending_condition(player: Dict[str, Any], condition: Dict[str, Any]) -> bool:
    """
    检查结局条件是否满足（所有条件需同时满足）。
    条件格式：
    {
        "属性名": {"gte": 数值, "lte": 数值},   # 属性比较
        "flags": {"标记名": True/False}        # 标记判断
    }
    特殊：condition 为空字典 {} 表示无条件满足。
    """
    for key, value in condition.items():
        if key == "flags":
            flags = player.get("flags", {})
            for flag_name, flag_value in value.items():
                if flags.get(flag_name) != flag_value:
                    return False
        else:
            # 属性判断
            if key not in player:
                return False
            actual = player[key]
            if isinstance(value, dict):
                if "gte" in value and actual < value["gte"]:
                    return False
                if "lte" in value and actual > value["lte"]:
                    return False
                if "eq" in value and actual != value["eq"]:
                    return False
            else:
                if actual != value:
                    return False
    return True
